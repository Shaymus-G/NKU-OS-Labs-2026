int kern_init(void)
{
    extern char edata[], end[];
    memset(edata, 0, end - edata);
    dtb_init();
    cons_init(); // init the console

    const char *message = "(THU.CST) os is loading ...";
    cprintf("%s\n\n", message);

    print_kerninfo();

    // grade_backtrace();

    pmm_init();

    cprintf("A isolated check: print current page table\n");
    print_pgdir();

    vmm_init();

    cprintf("A isolated check: PMM and VMM initialization returned\n");

    sbi_shutdown();

    while (1) {
    }
}
