===== make build =====
+ cc kern/init/entry.S
+ cc kern/init/init.c
+ cc kern/libs/stdio.c
+ cc kern/driver/console.c
+ cc libs/printfmt.c
+ cc libs/readline.c
+ cc libs/sbi.c
+ cc libs/string.c
+ ld bin/kernel
riscv64-unknown-elf-objcopy bin/kernel --strip-all -O binary bin/ucore.img

===== make -n qemu =====
qemu-system-riscv64 \
	-machine virt \
	-nographic \
	-bios default \
	-device loader,file=bin/ucore.img,addr=0x80200000

===== make -n debug =====
qemu-system-riscv64 \
	-machine virt \
	-nographic \
	-bios default \
	-device loader,file=bin/ucore.img,addr=0x80200000\
	-s -S

===== make -n gdb =====
riscv64-unknown-elf-gdb \
    -ex 'file bin/kernel' \
    -ex 'set arch riscv:rv64' \
    -ex 'target remote localhost:1234'
